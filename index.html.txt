    <script>
        // دالة تشغيل خرائط جوجل الحقيقية
        function initMap() {
            const muscat = { lat: 23.5880, lng: 58.3829 };
            const map = new google.maps.Map(document.getElementById("map"), {
                zoom: 12,
                center: muscat,
            });
            new google.maps.Marker({
                position: muscat,
                map: map,
                title: "مركز التوصيل الرئيسي - مسقط",
            });
        }

        // التبديل بين الأقسام
        function switchTab(tab) {
            ['menu', 'driver', 'restaurants', 'admin'].forEach(t => {
                document.getElementById('tab-' + t).classList.add('hidden');
                document.getElementById('btn-' + t).className = "py-2.5 text-gray-500 rounded-lg";
            });
            document.getElementById('tab-' + tab).classList.remove('hidden');
            document.getElementById('btn-' + tab).className = "py-2.5 text-red-600 bg-red-50 font-bold rounded-lg";
        }

        // سلة المشتريات
        let cart = [];
        function addToCart(name, price) {
            let item = cart.find(i => i.name === name);
            if (item) { item.qty++; } else { cart.push({ name, price, qty: 1 }); }
            updateCartUI();
            alert("تمت إضافة (" + name + ") للسلة!");
        }

        function updateCartUI() {
            let count = cart.reduce((sum, i) => sum + i.qty, 0);
            document.getElementById('cart-count').innerText = count;
            let container = document.getElementById('cart-items');
            if (cart.length === 0) {
                container.innerHTML = '<p class="text-gray-400 text-center py-4 text-xs">السلة فارغة حالياً</p>';
                document.getElementById('cart-total').innerText = "0.000 ر.ع";
                return;
            }
            let html = '';
            let total = 0;
            cart.forEach((item, index) => {
                let itemTotal = item.price * item.qty;
                total += itemTotal;
                html += `
                    <div class="flex justify-between items-center bg-gray-50 p-2.5 rounded-xl">
                        <div>
                            <p class="font-bold text-xs">${item.name}</p>
                            <p class="text-xs text-gray-500">${item.price} ر.ع × ${item.qty}</p>
                        </div>
                        <div class="flex items-center gap-2">
                            <span class="font-bold text-red-600 text-xs">${itemTotal.toFixed(3)} ر.ع</span>
                            <button onclick="removeFromCart(${index})" class="text-red-500 text-xs px-2 py-1 bg-red-50 rounded">حذف</button>
                        </div>
                    </div>
                `;
            });
            container.innerHTML = html;
            document.getElementById('cart-total').innerText = total.toFixed(3) + " ر.ع";
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCartUI();
        }

        function toggleCart() {
            document.getElementById('cart-modal').classList.toggle('hidden');
        }

        // إرسال الطلب وحفظه ليظهر في لوحة تحكم المندوب
        function checkoutOrder() {
            if (cart.length === 0) { alert("السلة فارغة!"); return; }
            let totalText = document.getElementById('cart-total').innerText;
            
            let newOrder = {
                id: Math.floor(1000 + Math.random() * 9000),
                items: [...cart],
                total: totalText,
                status: 'في انتظار قبول المندوب'
            };

            let orders = JSON.parse(localStorage.getItem('driverOrders') || '[]');
            orders.unshift(newOrder);
            localStorage.setItem('driverOrders', JSON.stringify(orders));

            cart = [];
            updateCartUI();
            toggleCart();
            alert("تم إرسال طلبك بنجاح! سيتم تنبيه مندوب التوصيل فوراً.");
            loadDriverOrders();
            switchTab('driver');
        }

        // تحميل وعرض الطلبات في لوحة تحكم المندوب
        function loadDriverOrders() {
            let orders = JSON.parse(localStorage.getItem('driverOrders') || '[]');
            let container = document.getElementById('driver-orders-list');
            
            if (orders.length === 0) {
                container.innerHTML = '<div class="text-center py-6 text-gray-400 text-sm">لا توجد طلبات جديدة حالياً</div>';
                return;
            }

            container.innerHTML = orders.map((order, idx) => `
                <div class="bg-gray-50 p-3 rounded-xl border space-y-2">
                    <div class="flex justify-between items-center border-b pb-2">
                        <span class="font-bold text-xs text-red-600">طلب رقم #${order.id}</span>
                        <span class="text-xs px-2 py-0.5 rounded font-bold ${
                            order.status.includes('قبول') ? 'bg-yellow-100 text-yellow-700' :
                            order.status.includes('بدء') ? 'bg-blue-100 text-blue-700' :
                            order.status.includes('اكتمل') ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'
                        }">${order.status}</span>
                    </div>
                    <div class="text-xs text-gray-600">
                        ${order.items.map(i => `• ${i.name} (×${i.qty})`).join('<br>')}
                        <div class="mt-1 font-bold text-gray-800">الإجمالي: ${order.total}</div>
                    </div>
                    
                    <div class="grid grid-cols-2 gap-2 pt-1">
                        <button onclick="updateOrderStatus(${idx}, 'تم قبول الطلب')" class="bg-blue-600 text-white py-1.5 rounded-lg text-xs font-bold">قبول الطلب</button>
                        <button onclick="updateOrderStatus(${idx}, 'تم رفض الطلب')" class="bg-red-600 text-white py-1.5 rounded-lg text-xs font-bold">رفض الطلب</button>
                        <button onclick="updateOrderStatus(${idx}, 'بدء العمل / التوصيل 🛵')" class="bg-amber-500 text-white py-1.5 rounded-lg text-xs font-bold col-span-2">بدء العمل والتوصيل</button>
                        <button onclick="updateOrderStatus(${idx}, 'اكتمل التوصيل بنجاح ✅')" class="bg-green-600 text-white py-1.5 rounded-lg text-xs font-bold col-span-2">تم التسليم بنجاح</button>
                    </div>
                </div>
            `).join('');
        }

        function updateOrderStatus(index, newStatus) {
            let orders = JSON.parse(localStorage.getItem('driverOrders') || '[]');
            orders[index].status = newStatus;
            localStorage.setItem('driverOrders', JSON.stringify(orders));
            loadDriverOrders();
        }

        // اشتراك المطاعم (سهل وسريع)
        document.getElementById('easy-subscription-form').addEventListener('submit', function(e) {
            e.preventDefault();
            let name = document.getElementById('sub-res-name').value;
            let phone = document.getElementById('sub-phone').value;
            
            let restaurants = JSON.parse(localStorage.getItem('subscribedRestaurants') || '[]');
            restaurants.push({ name, phone, plan: '30 ر.ع شهرياً' });
            localStorage.setItem('subscribedRestaurants', JSON.stringify(restaurants));

            alert("مبارك! تم تفعيل اشتراك مطعم (" + name + ") بنجاح.");
            this.reset();
            loadSubscribedRestaurants();
        });

        function loadSubscribedRestaurants() {
            let list = JSON.parse(localStorage.getItem('subscribedRestaurants') || '[]');
            let container = document.getElementById('active-restaurants-list');
            if (list.length === 0) {
                container.innerHTML = '<div class="text-xs text-gray-400">لا توجد مطاعم مسجلة بعد</div>';
                return;
            }
            container.innerHTML = list.map(r => `
                <div class="bg-white p-2.5 rounded-xl border flex justify-between items-center text-xs">
                    <div>
                        <span class="font-bold">${r.name}</span>
                        <span class="block text-gray-400">هاتف: ${r.phone}</span>
                    </div>
                    <span class="bg-green-100 text-green-700 px-2 py-1 rounded font-bold">مشترك نشط</span>
                </div>
            `).join('');
        }

        // إدارة الموظفين
        document.getElementById('employee-form').addEventListener('submit', function(e) {
            e.preventDefault();
            let name = document.getElementById('emp-name').value;
            let phone = document.getElementById('emp-phone').value;
            let list = JSON.parse(localStorage.getItem('employees') || '[]');
            list.push({ name, phone });
            localStorage.setItem('employees', JSON.stringify(list));
            alert("تمت إضافة الموظف بنجاح!");
            this.reset();
            loadEmployees();
        });

        function loadEmployees() {
            let emps = JSON.parse(localStorage.getItem('employees') || '[]');
            let container = document.getElementById('employees-list');
            container.innerHTML = emps.map(emp => `
                <div class="bg-white p-2.5 rounded-xl border flex justify-between items-center text-xs">
                    <div>
                        <span class="font-bold">${emp.name}</span>
                        <span class="block text-gray-400">${emp.phone}</span>
                    </div>
                    <span class="bg-blue-100 text-blue-700 px-2 py-1 rounded font-bold">مندوب</span>
                </div>
            `).join('');
        }

        // تحميل البيانات عند فتح التطبيق
        loadDriverOrders();
        loadSubscribedRestaurants();
        loadEmployees();
    </script>
